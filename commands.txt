# ==========================================================
# Federated Story Point Estimation - Command Runbook
# ==========================================================
# Notes:
# - Run these commands from PowerShell.
# - Edit paths only if your folder locations differ.
# - "--seed" keeps training reproducible.
# - Data dir must contain one CSV per project (including lsstcorp.csv).
# - Lsstcorp is used for warm-start pre-training; the remaining
#   projects become the FL clients.


# ==========================================================
# 1) TRAINING — Standard run (FedProx + FFA-LoRA + warm-start)
# ==========================================================
# What this run does:
# - Warm-starts LoRA + head on Lsstcorp (up to 10 epochs, early-stop on val macro-F1)
# - Federates across the remaining projects as clients
# - 20 global FL rounds, 1 local epoch per client per round
# - All clients participate each round (fraction = 1.0)
# - FFA-LoRA: only LoRA B matrices + head are trained and aggregated
# - Results (per-project CSV, markdown, confusion matrices) saved to --save-dir/results/

& "d:/Federated Learning using DL/.venv/Scripts/python.exe" "d:/Federated Learning using DL/train_federated_dl.py" `
	--data-dir "D:\Federated Learning using DL\data_to_train_on" `
	--device cuda `
	--model-name "distilbert-base-uncased" `
	--max-length 384 `
	--rounds 20 `
	--local-epochs 1 `
	--batch-size 8 `
	--lr 3e-5 `
	--weight-decay 1e-4 `
	--prox-mu 1e-2 `
	--clients-per-round-fraction 1.0 `
	--local-sample-ratio-per-epoch 1.0 `
	--lora-r 8 `
	--lora-alpha 16 `
	--lora-dropout 0.05 `
	--lora-target-modules q_lin v_lin `
	--warmstart-project lsstcorp `
	--warmstart-epochs 10 `
	--warmstart-patience 3 `
	--warmstart-lr 3e-5 `
	--save-dir "D:\Federated Learning using DL\artifacts" `
	--central-log-every 1 `
	--seed 42


# ==========================================================
# 2) TRAINING VARIANT — Partial client participation + local subsampling
# ==========================================================
# What changes vs command above:
# - 60% of clients selected each round
# - Each selected client trains on a fresh 50% random subset per local epoch


& "d:/Federated Learning using DL/.venv/Scripts/python.exe" "d:/Federated Learning using DL/train_federated_dl.py" `
	--data-dir "D:\Federated Learning using DL\data_to_train_on" `
	--device cuda `
	--skip-centralized `
	--model-name "distilbert-base-uncased" `
	--max-length 384 `
	--rounds 25 `
	--local-epochs 2 `
	--batch-size 8 `
	--lr 3e-5 `
	--weight-decay 1e-4 `
	--prox-mu 1e-3 `
	--clients-per-round-fraction 0.8 `
	--local-sample-ratio-per-epoch 0.75 `
	--lora-r 16 `
	--lora-alpha 32 `
	--lora-dropout 0.05 `
	--lora-target-modules q_lin v_lin `
	--warmstart-project Lsstcorp_Data_management `
	--warmstart-epochs 15 `
	--warmstart-patience 5 `
	--warmstart-lr 3e-5 `
	--save-dir "D:\Federated Learning using DL\artifacts_variant" `
	--central-log-every 1 `
	--seed 42


# ==========================================================
# 3) TRAINING VARIANT — Standard LoRA (FFA-LoRA off, A+B both trained)
# ==========================================================
# Use --no-ffa-lora to train and aggregate both A and B matrices.

& "d:/Federated Learning using DL/.venv/Scripts/python.exe" "d:/Federated Learning using DL/train_federated_dl.py" `
	--data-dir "D:\Federated Learning using DL\data_to_train_on" `
	--device cuda `
	--model-name "distilbert-base-uncased" `
	--max-length 384 `
	--rounds 20 `
	--local-epochs 1 `
	--batch-size 8 `
	--lr 3e-5 `
	--prox-mu 1e-2 `
	--lora-r 8 `
	--lora-alpha 16 `
	--lora-target-modules q_lin v_lin `
	--no-ffa-lora `
	--warmstart-project lsstcorp `
	--warmstart-epochs 10 `
	--warmstart-patience 3 `
	--save-dir "D:\Federated Learning using DL\artifacts_std_lora" `

# ==========================================================
